# M11 Grafiktest (docs/45 §6): Kamerahöhe beim Schleichen/Schwimmen, Sprinten über die Taste, Sicht unter Wasser und in Lava,
# keine Bewegung bei geöffnetem Inventar, niedrige Decke. Bildschirmfotos, Bewegungszustand (motion-dump), GL-Bilanz beim Beenden.
# Aufruf: bash analysis/tools/m6-run.sh analysis/tools/m11-gfx.txt <label> ["--width 1600 --height 900"]
server-console ${CONSOLE}
wait-play
mark joined
console gamerule spawn_mobs false
console forceload add -32 -64 47 15
console kill @e[type=!player]
console fill -14 99 -48 20 99 -16 minecraft:grass_block
console fill -14 100 -48 20 106 -16 minecraft:air
console fill -14 103 -44 20 103 -22 minecraft:light[level=15]
# Wasserbecken (3 tief) x 4..8, z -40..-34; Lavabecken (2 tief) x -10..-8, z -40..-38; niedrige Decke x -4..-2, z -30..-26
console fill 3 95 -41 9 99 -33 minecraft:stone
console fill 4 96 -40 8 99 -34 minecraft:water
console fill -11 96 -41 -7 99 -37 minecraft:stone
console fill -10 98 -40 -8 99 -38 minecraft:lava
console fill -4 101 -30 -2 101 -26 minecraft:oak_slab[type=top]
console gamemode survival ${NAME}
console effect clear ${NAME}
console effect give ${NAME} minecraft:saturation 1 20 true
console tp ${NAME} 0.5 100 -27.5 180 0
wait 3
wait-meshes
motion-dump stand
screenshot ${OUT}/60-stand.png
# --- Schleichen: Pose und geglättete Kamerahöhe (1,62 → 1,27)
hold s
wait 1
motion-dump sneak
screenshot ${OUT}/61-sneak.png
hold -
wait 1
motion-dump unsneak
# --- Sprinten über die Sprinttaste
hold fp
wait 1.5
motion-dump sprint
screenshot ${OUT}/62-sprint.png
hold -
wait 1.5
# --- Wasser: Eintauchen, Sicht (waterVisionTime), Schwimmen
console tp ${NAME} 6.5 100 -32.5 180 0
wait 2
hold f
wait 1.5
hold -
wait 2
motion-dump underwater
screenshot ${OUT}/63-underwater.png
wait 6
motion-dump underwater-vision
screenshot ${OUT}/64-underwater-vision.png
look 180 20
hold fp
wait 2
motion-dump swim
screenshot ${OUT}/65-swim.png
hold -
look 180 -40
hold j
wait 3
hold -
# --- Lava mit Feuerresistenz (Sicht 0..5)
console effect give ${NAME} minecraft:fire_resistance 60 0 true
console tp ${NAME} -8.5 98 -38.5 180 0
wait 2
motion-dump lava
screenshot ${OUT}/66-lava.png
console tp ${NAME} 0.5 100 -27.5 180 0
wait 2
console effect clear ${NAME}
# --- Inventar offen: Tasten ohne Wirkung
motion-dump before-inventory
inventory open
wait 0.5
keys fp 1.5
motion-dump inventory-keys
inventory close
wait 0.5
# --- niedrige Decke (1,5 Blöcke): Pose bleibt Schleichen ohne Shift
console tp ${NAME} -2.5 100 -28.5 180 0
wait 2
motion-dump low-ceiling
screenshot ${OUT}/67-low-ceiling.png
console tp ${NAME} 0.5 100 -27.5 180 0
wait 2
console fill 3 95 -41 9 99 -33 minecraft:grass_block
console fill -11 95 -41 -7 99 -37 minecraft:grass_block
console fill -4 101 -30 -2 101 -26 minecraft:air
console fill -14 103 -44 20 103 -22 minecraft:air
console gamerule spawn_mobs true
console gamemode creative ${NAME}
wait 1
mark done
